直觉与类比

类比 · 三个工作台轮流用

F2 的小组只有一张工作台：搬料时冲压机停，冲压时叉车停。F3 有三张：冲压机在用 1 号台的料时，2 号台的料已经到了，3 号台正在卸货。每用完一张台子，就让叉车去给它装“三轮之后”要用的料。工人从台子上取料也不再一件件拿，而是一次抱一摞（ldmatrix.x4），而且在冲这一摞的同时，先把下一摞抱到手边（寄存器双缓冲）。

图解：一圈主循环里发生了什么

逐步演示：用 trace(5, 3) 走一遍

▶ k_tiles = 5，STAGES = 3
1. 预取 STAGES − 1 = 2 级

load(0 → slot 0)、load(1 → slot 1)，每次一个 commit group。count 从 5 减到 3，k_next 走到 2。

2. 等第 0 级

cp_async_wait<1>()：允许最近的 1 组还在飞，所以第 0 组（tile 0）一定到了。__syncthreads() 之后读 tile 0 的第一个 k16 进寄存器。

3. 第 0 圈

kb = 0：先发 load(2 → slot 2)，再 write = 0、read = 1；然后算 tile 0（fragment 来自 slot 0）。kb = 3：wait<1> 确认 tile 1 到齐，切到 slot 1。

4. 第 1、2 圈

第 1 圈 load(3 → slot 0)——slot 0 里的 tile 0 刚在第 0 圈用完，第 0 圈末尾的 __syncthreads() 保证所有 warp 都读完了；算 tile 1。第 2 圈 load(4 → slot 1)，算 tile 2。

5. 第 3、4 圈：尾巴上的多余 load

count 已经 ≤ 0，k_next 停在 4，所以又发了两次 load(4)（写进 slot 2、slot 0）。它们不会再被读，多余但无害——这正是 redundant_loads = STAGES − 1。第 3 圈算 tile 3（slot 0），第 4 圈算 tile 4（slot 1）。一共 5 圈 = k_tiles。

为什么 kb == 0 时就可以覆盖 write 那一级，不怕别的 warp 还在读？

write 那一级就是上一圈刚用完的那一级（write = read 之前的 read）。上一圈 kb == KB−1 处有一个

__syncthreads() ，在那之后所有 warp 读的都是新的 read 级——旧的那一级已经没人读了。这个 barrier 同时承担了“下一级到齐”和“旧的一级可以覆盖”两件事，这就是为什么一圈只需要一个 barrier。

"""F3 · 多级 cp.async 流水线 + ldmatrix + 寄存器双缓冲：sgemm_sm80 的完整结构 — reference solution.

GPU 部分在 `f3_pipelined_gemm.cu`。主循环里最容易写错的是“哪一级 SMEM 在什么时候被写、什么时候被读”。
本文件把 .cu 里的循环（逐行照抄 CUTLASS sgemm_sm80.cu 的 L132–L305）用整数模拟一遍，得到完整的事件序列，
再检查它的不变量：每个 k-tile 恰好被算一次、按顺序、算的时候那一级 SMEM 里确实是它，而且已经 wait 过。

.cu 里的循环（S = STAGES，KB = BK / 16 = 4 个 k16 步）
    prefetch:  for s in 0..S-2:  load(k_next → slot s); commit; count -= 1; if count > 0: k_next += 1
    read = 0, write = S-1
    wait(S-2); sync                                   # 第 0 级到齐
    while count > -(S-1):                             # 一共转 k_tiles 圈
        for kb in 0..KB-1:
            if kb == KB-1: wait(S-2); sync            # 切到下一级之前，确认它到齐
            （读下一个 k16 的 fragment）
            if kb == 0:
                load(k_next → slot write); commit; count -= 1; if count > 0: k_next += 1
                write = read; read = (read + 1) % S
            compute(这一圈的 k-tile, kb)
- 一开始 count = k_tiles。k_next 在 count 降到 0 以后不再前进，所以最后 S-1 次 load 重复加载最后一个 k-tile
  （多余但无害：它写进的 slot 不会再被读）。
- 第 i 圈（i = 0..k_tiles-1）算的是第 i 个 k-tile，它在 slot i % S 里。
  注意：第 i 圈的 kb = 0 那一步先发了下一个 load 并把 read 往前推，但这一圈的 kb = 0..KB-1 用的
  fragment 是在上一圈末尾（或 prologue）从“旧的 read”那一级读出来的——这就是寄存器双缓冲。

trace(k_tiles, stages) 返回事件列表，每个事件是一个 tuple：
    ("load", k_tile, slot)       发出一次 cp.async（一个 commit group）
    ("wait", pending_allowed)    cp_async_wait<pending_allowed>() + __syncthreads()
    ("compute", k_tile, slot)    这一圈的第一个 k16 步（每圈只记一次）
规则：wait(n) 之后，除了最近的 n 个 group，之前发出的 group 都已经落地。

check_trace(events, stages)：重放事件，检查
    1. compute 的 k_tile 是 0, 1, 2, … 依次出现，每个一次；
    2. compute 时 slot 里最后一次 load 的就是这个 k_tile，并且那次 load 已经被某个 wait 确认落地；
    3. 一次 load 写进的 slot，不能是“已加载、尚未被 compute 用过”的那个 k_tile 所在的 slot（否则覆盖了没用过的数据）。
   全部满足返回 True，否则抛 AssertionError（消息写明哪一条）。

其他
    smem_bytes(BM, BN, BK, stages) = stages·(BM + BN)·BK·2
    max_stages(BM, BN, BK, limit=99·1024) = 放得下的最大 stage 数（至少 1，否则 ValueError）
    wait_arg(stages) = stages - 2                 cp_async_wait 的参数：允许还在飞的 group 数
    inflight_bytes(BM, BN, BK, stages) = (stages - 1)·(BM + BN)·BK·2   稳态时同时在路上的字节数
    redundant_loads(k_tiles, stages) = 总 load 次数 − k_tiles

API
    trace(k_tiles, stages) -> list[tuple]
    check_trace(events, stages) -> bool
    smem_bytes, max_stages, wait_arg, inflight_bytes, redundant_loads

from __future__ import annotations


def trace(k_tiles: int, stages: int) -> list:
    if k_tiles < 1 or stages < 2:
        raise ValueError("need k_tiles >= 1 and stages >= 2")
    ev = []
    count, k_next = k_tiles, 0
    for s in range(stages - 1):
        ev.append(("load", k_next, s))
        count -= 1
        if count > 0:
            k_next += 1
    read, write = 0, stages - 1
    ev.append(("wait", stages - 2))
    i = 0
    while count > -(stages - 1):
        compute_slot = read            # 这一圈用的 fragment 来自 kb = 0 之前的 read
        # kb == 0：先发下一级 load，再推进 read / write
        ev.append(("load", k_next, write))
        count -= 1
        if count > 0:
            k_next += 1
        write, read = read, (read + 1) % stages
        ev.append(("compute", i, compute_slot))
        # kb == KB-1：切到下一级之前等它到齐
        ev.append(("wait", stages - 2))
        i += 1
    return ev


def check_trace(events: list, stages: int) -> bool:
    slot_tile = {}        # slot -> 最后一次 load 进来的 k_tile
    slot_group = {}       # slot -> 那次 load 的 group 编号
    groups = 0
    landed = -1           # 编号 <= landed 的 group 都已落地
    next_compute = 0
    loaded_unused = set()
    for e in events:
        if e[0] in ("load", "compute") and not 0 <= e[2] < stages:
            raise AssertionError(f"slot {e[2]} outside [0, {stages})")
        if e[0] == "load":
            _, kt, slot = e
            if slot in slot_tile and slot_tile[slot] in loaded_unused:
                raise AssertionError(f"rule 3: load of k-tile {kt} overwrites slot {slot} holding unused k-tile {slot_tile[slot]}")
            slot_tile[slot], slot_group[slot] = kt, groups
            groups += 1
            if kt >= next_compute:
                loaded_unused.add(kt)
        elif e[0] == "wait":
            landed = max(landed, groups - 1 - e[1])
        elif e[0] == "compute":
            _, kt, slot = e
            if kt != next_compute:
                raise AssertionError(f"rule 1: computed k-tile {kt}, expected {next_compute}")
            if slot_tile.get(slot) != kt:
                raise AssertionError(f"rule 2: slot {slot} holds {slot_tile.get(slot)}, not k-tile {kt}")
            if slot_group[slot] > landed:
                raise AssertionError(f"rule 2: k-tile {kt} computed before its cp.async group was waited for")
            loaded_unused.discard(kt)
            next_compute += 1
    return True


def smem_bytes(BM: int, BN: int, BK: int, stages: int) -> int:
    return stages * (BM + BN) * BK * 2


def max_stages(BM: int, BN: int, BK: int, limit: int = 99 * 1024) -> int:
    s = limit // ((BM + BN) * BK * 2)
    if s < 1:
        raise ValueError(f"even one stage of {BM}x{BN}x{BK} does not fit in {limit} bytes")
    return s


def wait_arg(stages: int) -> int:
    return stages - 2


def inflight_bytes(BM: int, BN: int, BK: int, stages: int) -> int:
    return (stages - 1) * (BM + BN) * BK * 2


def redundant_loads(k_tiles: int, stages: int) -> int:
    return sum(1 for e in trace(k_tiles, stages) if e[0] == "load") - k_tiles

constexpr int kSmemBytes = Cfg::SmemBytes;

// ------------------------------------------------------------------ kernel
__global__ __launch_bounds__(kThreads) void gemm_kernel(int M, int N, int K, TA const* A, TB const* B, TC* C) {
  static_assert(size(TiledMma{}) == kThreads && size(G2SCopy{}) == kThreads, "thread counts must match");

  // ① 全局张量和这个 CTA 的 tile（A1、B1）
  Tensor mA = make_tensor(make_gmem_ptr(A), make_shape(M, K), make_stride(K, Int<1>{}));   // (M,K)
  Tensor mB = make_tensor(make_gmem_ptr(B), make_shape(N, K), make_stride(K, Int<1>{}));   // (N,K)
  Tensor mC = make_tensor(make_gmem_ptr(C), make_shape(M, N), make_stride(N, Int<1>{}));   // (M,N)
  auto cta_tiler = make_shape(BM{}, BN{}, BK{});
  auto cta_coord = make_coord(blockIdx.x, blockIdx.y, _);
  Tensor gA = local_tile(mA, cta_tiler, cta_coord, Step<_1, X, _1>{});   // (BM,BK,k)
  Tensor gB = local_tile(mB, cta_tiler, cta_coord, Step<X, _1, _1>{});   // (BN,BK,k)
  Tensor gC = local_tile(mC, cta_tiler, cta_coord, Step<_1, _1, X>{});   // (BM,BN)

  // ② SMEM 环形缓冲（A3）
  extern __shared__ __align__(128) unsigned char smem_raw[];
  TA* smemA = reinterpret_cast<TA*>(smem_raw);
  TB* smemB = smemA + cosize_v<SmemLayoutA>;
  Tensor sA = make_tensor(make_smem_ptr(smemA), SmemLayoutA{});   // (BM,BK,STAGES)
  Tensor sB = make_tensor(make_smem_ptr(smemB), SmemLayoutB{});   // (BN,BK,STAGES)

  // ③ GMEM → SMEM 的线程分工（A2）
  G2SCopy g2s;
  auto thr_g2s = g2s.get_slice(threadIdx.x);
  Tensor tAgA = thr_g2s.partition_S(gA);   // (CPY,CPY_M,CPY_K,k)
  Tensor tAsA = thr_g2s.partition_D(sA);   // (CPY,CPY_M,CPY_K,STAGES)
  Tensor tBgB = thr_g2s.partition_S(gB);
  Tensor tBsB = thr_g2s.partition_D(sB);

  // ④ MMA 分工和累加器（A4）
  TiledMma mma;
  auto thr_mma = mma.get_slice(threadIdx.x);
  Tensor tCgC = thr_mma.partition_C(gC);                  // (MMA,MMA_M,MMA_N)
  Tensor tCrA = thr_mma.partition_fragment_A(sA(_, _, 0));  // (MMA,MMA_M,MMA_K) 寄存器
  Tensor tCrB = thr_mma.partition_fragment_B(sB(_, _, 0));  // (MMA,MMA_N,MMA_K)
  Tensor tCrC = thr_mma.make_fragment_C(tCgC);            // (MMA,MMA_M,MMA_N) FP32 累加器
  clear(tCrC);

  // ⑤ SMEM → 寄存器：ldmatrix，按 MMA 的 A/B 布局“重排”出来的 TiledCopy
  auto s2r_a = make_tiled_copy_A(S2RAtom{}, mma);
  auto thr_s2r_a = s2r_a.get_slice(threadIdx.x);
  Tensor tXsA = thr_s2r_a.partition_S(sA);   // (CPY,MMA_M,MMA_K,STAGES)
  Tensor tXrA = thr_s2r_a.retile_D(tCrA);    // (CPY,MMA_M,MMA_K)
  auto s2r_b = make_tiled_copy_B(S2RAtom{}, mma);
  auto thr_s2r_b = s2r_b.get_slice(threadIdx.x);
  Tensor tXsB = thr_s2r_b.partition_S(sB);
  Tensor tXrB = thr_s2r_b.retile_D(tCrB);

  // [TODO ⑥⑦] 预取 STAGES-1 级；等第 0 级到齐并读第一个 k16；主循环（kb==KB-1 时切换 read 级并 wait、kb==0 时发下一级 load 并推进 read/write、每步先读 kb+1 的 fragment 再算 kb）
  // ⑥ 预取：先把前 STAGES-1 个 k-tile 发出去，每个一组（commit group）
  int k_tile_count = size<3>(tAgA);
  int k_tile_next = 0;
  CUTE_UNROLL
  for (int s = 0; s < kStages - 1; ++s) {
    copy(g2s, tAgA(_, _, _, k_tile_next), tAsA(_, _, _, s));
    copy(g2s, tBgB(_, _, _, k_tile_next), tBsB(_, _, _, s));
    cp_async_fence();
    --k_tile_count;
    if (k_tile_count > 0) ++k_tile_next;
  }

  int smem_read = 0;
  int smem_write = kStages - 1;
  Tensor tXsA_p = tXsA(_, _, _, smem_read);
  Tensor tXsB_p = tXsB(_, _, _, smem_read);
  constexpr int kBlocks = decltype(size<2>(tCrA))::value;   // BK / 16 = 4 个 k16 步

  // 等第 0 级到齐，先把它的第一个 k16 读进寄存器
  cp_async_wait<kStages - 2>();
  __syncthreads();
  copy(s2r_a, tXsA_p(_, _, Int<0>{}), tXrA(_, _, Int<0>{}));
  copy(s2r_b, tXsB_p(_, _, Int<0>{}), tXrB(_, _, Int<0>{}));

  // ⑦ 主循环：每个 k-tile 4 个 k16 步；第 0 步发下一级加载，最后一步切换到下一级 SMEM
  CUTE_NO_UNROLL
  while (k_tile_count > -(kStages - 1)) {
    CUTE_UNROLL
    for (int kb = 0; kb < kBlocks; ++kb) {
      if (kb == kBlocks - 1) {
        tXsA_p = tXsA(_, _, _, smem_read);
        tXsB_p = tXsB(_, _, _, smem_read);
        cp_async_wait<kStages - 2>();
        __syncthreads();
      }
      int kb_next = (kb + 1) % kBlocks;
      copy(s2r_a, tXsA_p(_, _, kb_next), tXrA(_, _, kb_next));   // 寄存器双缓冲：先读下一步
      copy(s2r_b, tXsB_p(_, _, kb_next), tXrB(_, _, kb_next));
      if (kb == 0) {
        copy(g2s, tAgA(_, _, _, k_tile_next), tAsA(_, _, _, smem_write));
        copy(g2s, tBgB(_, _, _, k_tile_next), tBsB(_, _, _, smem_write));
        cp_async_fence();
        --k_tile_count;
        if (k_tile_count > 0) ++k_tile_next;
        smem_write = smem_read;
        smem_read = (smem_read == kStages - 1) ? 0 : smem_read + 1;
      }
      gemm(mma, tCrA(_, _, kb), tCrB(_, _, kb), tCrC);           // 再算这一步
    }
  }

  // [/TODO]

  // ⑧ epilogue：累加器直接写回（FP32；F4 会改成 BF16 + 谓词）
  copy(tCrC, tCgC);
}

// ------------------------------------------------------------------ host
inline void launch(int M, int N, int K, TA const* A, TB const* B, TC* C, cudaStream_t stream = 0) {
  dim3 grid(M / BM::value, N / BN::value);
  gemm_kernel<<<grid, kThreads, kSmemBytes, stream>>>(M, N, K, A, B, C);
}

inline void setup() {
  CUDA_CHECK(cudaFuncSetAttribute(gemm_kernel, cudaFuncAttributeMaxDynamicSharedMemorySize, kSmemBytes));
}

}  // namespace f3

int main(int argc, char** argv) {
  auto a = drill::parse(argc, argv);
  if (a.M % 128 || a.N % 128 || a.K % 64) {
    std::printf("{\"ok\": false, \"error\": \"F3 needs M,N %% 128 == 0 and K %% 64 == 0\"}\n");
    return 1;
  }
  std::vector<__nv_bfloat16> hA(size_t(a.M) * a.K), hB(size_t(a.N) * a.K);
  drill::fill_bf16(hA, 1);
  drill::fill_bf16(hB, 2);
  auto* dA = drill::to_device(hA);
  auto* dB = drill::to_device(hB);
  float* dC = nullptr;
  CUDA_CHECK(cudaMalloc(&dC, size_t(a.M) * a.N * sizeof(float)));

  f3::setup();
  auto go = [&] { f3::launch(a.M, a.N, a.K, reinterpret_cast<f3::TA const*>(dA), reinterpret_cast<f3::TB const*>(dB), dC); };
  go();
  CUDA_CHECK(cudaDeviceSynchronize());
  std::vector<float> got(size_t(a.M) * a.N);
  CUDA_CHECK(cudaMemcpy(got.data(), dC, got.size() * sizeof(float), cudaMemcpyDeviceToHost));
  double err = drill::max_rel_err(got, drill::reference(a.M, a.N, a.K, dA, dB));
  float ms = drill::time_ms(go, a.iters);
  char extra[128];
  std::snprintf(extra, sizeof extra, ", \"smem_bytes\": %d, \"threads\": %d, \"stages\": %d", f3::kSmemBytes,
                f3::kThreads, f3::kStages);
  drill::report("f3_pipelined_128x128x64_s3", a, err, ms, extra);
  return 0;
}
